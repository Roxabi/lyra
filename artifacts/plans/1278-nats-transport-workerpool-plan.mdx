## Bootstrap Context

The 3-layer DI composition is mechanical. Each factory constructs stack bottom-up and tears down top-down:

**`bootstrap/factory/llm_overlay.py`** (replaces single-class construction):
```python
transport = NatsTransport(nc, default_timeout=timeout)
pool = WorkerPoolClient(
    transport,
    hb_subject=SUBJECTS.heartbeat,
    validate_worker_id=validate_worker_id,
)
await pool.start(nc)   # subscribes to hb_subject, spawns _heartbeat_loop
codec = LlmCodec()
client = LlmClient(pool, codec)
# teardown: await pool.stop()
```

**`bootstrap/factory/voice_overlay.py`** (same pattern × 3):
```python
# TTS
tts_transport = NatsTransport(nc, default_timeout=tts_timeout)
tts_pool = WorkerPoolClient(tts_transport, hb_subject=SUBJECTS.tts_heartbeat,
                             validate_worker_id=voice_validate_worker_id)
await tts_pool.start(nc)
tts_client = TtsClient(tts_pool, TtsCodec())

# STT — same pattern with SUBJECTS.stt_heartbeat
# Image — same pattern with SUBJECTS.image_heartbeat
```

**`bootstrap/factory/hub_builder.py`** and **`wiring_helpers.py`**: only type annotations change. `NatsLlmClient | None` → `LlmClient | None`. No logic changes. `VoiceBundle.nats_llm_client` field type updated accordingly in `wiring_helpers.py`.

Heartbeat lifecycle previously inherited from `NatsDriverBase` (subscribe on `start()`, cancel task + unsubscribe on `stop()`) is replayed inside `WorkerPoolClient.start/stop` with identical atomicity semantics. The defense-in-depth dual validation from `_worker_client_base.py` (`VALIDATE_WORKER_ID` + `registry.validate_nats_token` secondary guard) is preserved in `WorkerPoolClient._on_heartbeat`.

**T03** — Implement `NatsTransport.call` + `_sanitize` with all 4 error codes

File: `src/lyra/transport/nats_request_response.py`

```python
class NatsTransport:
    def __init__(self, nc: "NATS", *, default_timeout: float = 120.0) -> None:
        self._nc = nc
        self._default_timeout = default_timeout

    async def call(
        self, subject: str, payload: bytes, *, timeout: float | None = None
    ) -> "Result[bytes, SanitizedError]":
        t = timeout if timeout is not None else self._default_timeout
        try:
            reply = await self._nc.request(subject, payload, timeout=t)
            return Ok(reply.data)
        except TimeoutError as exc:
            return Err(self._sanitize(exc, context="call.timeout"))
        except NoRespondersError as exc:
            return Err(self._sanitize(exc, context="call.no_responders"))
        except nats.errors.Error as exc:
            return Err(self._sanitize(exc, context="call.error",
                                      payload_kb=len(payload)/1024))

    def _sanitize(self, exc: Exception, *, context: str = "",
                  payload_kb: float = 0.0) -> "SanitizedError":
        name = type(exc).__name__
        if "max_payload" in str(exc).lower():
            return SanitizedError(code="transport.payload_too_large",
                                  message=name, retryable=False)
        code_map = {"TimeoutError": "transport.timeout",
                    "NoRespondersError": "transport.no_responders"}
        code = code_map.get(name, "transport.error")
        return SanitizedError(code=code, message=name, retryable=True)
```

**T04** — Implement `NatsTransport.open_inbox` async context manager + `InboxStream`

File: `src/lyra/transport/nats_request_response.py`

```python
from contextlib import asynccontextmanager

@asynccontextmanager
async def open_inbox(self):
    inbox = self._nc.new_inbox()
    sub = await self._nc.subscribe(inbox)
    log.info("transport.inbox_open inbox=%s", inbox)
    async def _messages():
        try:
            while True:
                try:
                    msg = await sub.next_msg(timeout=self._default_timeout)
                    yield Ok(msg.data)
                except (TimeoutError, asyncio.TimeoutError) as exc:
                    yield Err(self._sanitize(exc, context="inbox.timeout"))
                    return
        except Exception as exc:
            yield Err(self._sanitize(exc, context="inbox.error"))
    try:
        yield InboxStream(inbox_subject=inbox, messages=_messages())
    finally:
        await sub.unsubscribe()
        log.info("transport.inbox_close inbox=%s", inbox)
```

```python
class WorkerPoolClient:
    def __init__(self, transport: NatsTransport, *, hb_subject: str,
                 validate_worker_id: Callable[[str], None],
                 hb_ttl: float = 15.0, name: str = "pool") -> None:
        self._transport = transport
        self._registry: WorkerRegistry = WorkerRegistry()
        self._cb: NatsCircuitBreaker = NatsCircuitBreaker()
        self._hb_subject = hb_subject
        self._validate_worker_id = validate_worker_id
        self._hb_ttl = hb_ttl
        self._name = name
        self._sub: object = None
        self._task: asyncio.Task | None = None

    async def start(self, nc: "NATS") -> None:
        self._sub = await nc.subscribe(self._hb_subject, cb=self._on_heartbeat)
        self._task = asyncio.create_task(self._heartbeat_loop(), name=f"{self._name}.hb")

    async def stop(self) -> None:
        if self._task: self._task.cancel()
        if self._sub: await self._sub.unsubscribe()
```

```python
async def request_with_routing(
    self, subject_fn: Callable[[str], str], payload: bytes, *,
    timeout: float | None = None, max_attempts: int | None = None
) -> "Result[bytes, SanitizedError]":
    if self._cb.is_open():
        return Err(SanitizedError(code="pool.circuit_open", message="CircuitOpen",
                                  retryable=True))
    attempt = 0
    for worker in self._registry.scored():
        if max_attempts is not None and attempt >= max_attempts:
            break
        subject = subject_fn(worker.id)
        log.info("pool.routing", extra={"pool": self._name, "worker_id": worker.id,
                  "subject": subject, "attempt": attempt, "result": "pending"})
        result = await self._transport.call(subject, payload, timeout=timeout)
        attempt += 1
        if isinstance(result, Ok):
            self._cb.record_success()
            return result
        if result.error.code in {"transport.timeout", "transport.no_responders"}:
            self._registry.mark_stale(worker.id)
        self._cb.record_failure()
```

```python
class LlmClient:
    capabilities: dict = {"streaming": True, "auth": "nats"}

    def __init__(self, pool: WorkerPoolClient, codec: LlmCodec,
                 *, timeout: float | None = None) -> None:
        self._pool = pool
        self._codec = codec
        self._timeout = timeout

    def is_alive(self, pool_id: str) -> bool:
        del pool_id  # LlmProvider protocol slot; pool is stateless per-pool-id
        return self._pool.is_pool_alive()

    async def complete(self, pool_id, text, model_cfg, system_prompt,
                       *, messages=None) -> LlmResult:
        payload, trace_id = self._codec.encode(text, model_cfg, system_prompt,
                                               messages, stream=False)
        result = await self._pool.request_with_routing(
            lambda _: SUBJECTS.generate_request, payload, max_attempts=1,
            timeout=self._timeout)
        return self._codec.decode(result, trace_id)

    async def stream(self, pool_id, text, model_cfg, system_prompt,
                     *, messages=None) -> AsyncIterator[LlmEvent]:
        payload, _ = self._codec.encode(text, model_cfg, system_prompt,
                                        messages, stream=True)
        return self._pool.stream_request(payload, timeout=self._timeout)
```
